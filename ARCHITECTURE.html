Data skel — convention de nommage des champs

codeAppscheme_field + ucfirst(codeAppscheme) → nom du champ stocké. Calculé via MongoCompat::toFieldName($code,$table).

// appscheme
{ codeAppscheme: "produit", libelle: "Produit" }

// appscheme_field
{ codeAppschemeField: "nom", libelle: "Nom" }

// appscheme_has_field (liaison)
{ codeAppscheme: "produit",
  codeAppschemeField: "nom",
  idFieldType: "text",
  idFieldGroup: "identification" }
// → document "produit" (collection réelle)
{ idproduit: 1042,
  nomProduit: "Vis M6",        // nom + ucfirst(produit)
  prixProduit: 3.5,
  dateCreationProduit: ISODate("2026-01-10"),
  idagentCreation: 7 }
